async function fetchRSS(url, sourceName) { const proxies = [ "https://api.rss2json.com/v1/api.json?rss_url=" + encodeURIComponent(url), "https://api.allorigins.win/raw?url=" + encodeURIComponent(url) ]; // 1. المحاولة الأولى عبر rss2json try { const response = await fetch(proxies[0]); if (response.ok) { const data = await response.json(); if (data.status === "ok" && Array.isArray(data.items) && data.items.length) { return data.items.map(item => ({ title: item.title, url: item.link, source: sourceName, category: classify(item.title), date: item.pubDate, is_urgent: /عاجل|breaking|urgent/i.test(item.title) })); } } } catch (e) {} // 2. محاولة احتياطية بقراءة XML مباشرة عبر proxy try { const res = await fetch(proxies[1]); if (res.ok) { const xmlText = await res.text(); const doc = new DOMParser().parseFromString(xmlText, "text/xml"); const entries = Array.from(doc.querySelectorAll("item, entry")); return entries.map(el => { const title = el.querySelector("title")?.textContent?.trim() || ""; const linkEl = el.querySelector("link"); const link = linkEl?.getAttribute("href") || linkEl?.textContent?.trim() || ""; const pubDate = el.querySelector("pubDate, updated, published")?.textContent?.trim() || new Date().toISOString(); return { title, url: link, source: sourceName, category: classify(title), date: pubDate, is_urgent: /عاجل|breaking|urgent/i.test(title) }; }).filter(item => item.title && item.url); } } catch (e) {} return []; }
